# ============================================================================
# uart_test -- ESP32-S3 UART1 发送自检固件（⭐ 一次性诊断用）
# ============================================================================
#
# ⭐ 目的：在【不依赖水池/起播】的前提下，独立验证
#     「ESP32 的 UART1（GPIO17 TX）能不能正确发出 9 字节帧」。
#
# 🔴 为什么要单独做这个：
#     正式固件（esp32/ble）里 `pool_start()` 【零调用】⇒ 放水路径没接上
#     ⇒ GPIO17 永远不发数据 ⇒ 直接拿正式固件验 UART 会【一片空白】，
#        然后白白去怀疑接线/波特率/共地。
#     ⭐ 本固件把 UART 单独隔离出来验，验完即弃（⛔ 不改正式固件）。
#
# ⭐ 行为：上电后每 1 秒从 GPIO17 发 36 帧（324 字节）测试帧，
#     同时 UART0 打日志说明发了什么。
#
# ⚠️ 接线（交叉！）：
#     ESP32 GPIO17 (TX) ──→ USB-TTL RX
#     ESP32 GPIO18 (RX) ←── USB-TTL TX   （本固件不用 RX，可不接）
#     ESP32 GND ─────────── USB-TTL GND  🔴 必须共地
#     ⛔ USB-TTL 的 5V/VCC 不要接
# ============================================================================

cmake_minimum_required(VERSION 3.16)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(uart_test)
